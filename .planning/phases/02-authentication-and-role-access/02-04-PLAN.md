**1. Add validateSessionQR mutation to convex/auth.ts:**

This mutation validates QR codes displayed at the venue for service users to join sessions.

```typescript
// Add to existing convex/auth.ts
export const validateSessionQR = mutation({
  args: { qrCode: v.string() },
  handler: async (ctx, { qrCode }) => {
    // QR format for sessions: "session:{sessionId}"
    if (!qrCode.startsWith("session:")) {
      throw new Error("Invalid session QR code");
    }
    const sessionId = qrCode.replace("session:", "") as Id<"sessions">;

    const session = await ctx.db.get(sessionId);
    if (!session) {
      throw new Error("Session not found");
    }
    if (!session.isActive) {
      throw new Error("This session has ended");
    }

    return {
      sessionId: session._id,
      location: session.location,
      date: session.date,
    };
  },
});
```
