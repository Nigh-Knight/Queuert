```typescript
const renderQueue = () => {
  // Prepare data with id field for Sortable
  const sortableData = filteredQueue.map(item => ({
    ...item,
    id: item._id,
  }));

  const handleDrop = ({ allPositions, movedItem }: { allPositions: { id: string; position: number }[], movedItem: any }) => {
    // Find the new position of the moved item
    const newPos = allPositions.find(p => p.id === movedItem.id)?.position;
    if (newPos === undefined) return;

    const userName = `${movedItem.user?.firstName || ''} ${movedItem.user?.lastName || ''}`.trim();

    // Show confirmation dialog
    setPendingReorder({
      userName: userName || 'Unknown User',
      newPosition: newPos + 1, // Convert to 1-indexed for display
      orderedIds: allPositions.sort((a, b) => a.position - b.position).map(p => p.id),
    });
  };

  const handleReorderConfirm = async () => {
    if (!pendingReorder || !sessionId) return;
    setIsActionLoading(true);
    try {
      await reorderMutation({
        sessionId,
        orderedIds: pendingReorder.orderedIds as Id<'queue'>[],
      });
      setPendingReorder(null);
    } catch (error) {
      Alert.alert('Error', 'Failed to reorder queue. Please try again.');
    } finally {
      setIsActionLoading(false);
    }
  };

  return (
    <View style={styles.queueContainer}>
      <SearchBar ... />

      {filteredQueue.length === 0 ? (
        // Empty state
      ) : (
        <Sortable
          data={sortableData}
          onDrop={handleDrop}
          renderItem={(item) => (
            <SortableItem id={item.id}>
              <QueueCard
                position={item.position}
                userName={...}
                ...
              />
            </SortableItem>
          )}
        />
      )}

      <FloatingActionButton onPress={handleAddUser} position="bottom-right" />

      <ReorderConfirmationModal
        visible={pendingReorder !== null}
        userName={pendingReorder?.userName || ''}
        newPosition={pendingReorder?.newPosition || 0}
        onClose={() => setPendingReorder(null)}
        onConfirm={handleReorderConfirm}
        isLoading={isActionLoading}
      />
    </View>
  );
};
```
